### Fix 7 — `@` file trigger without backend

In `CommandPanel.tsx`, the `@` trigger currently calls `api.scanWorkspace()` which hits the backend.
Replace with a local filesystem scan:

```typescript
import { readdirSync, statSync } from 'fs';
import { join, relative } from 'path';

function scanLocalFiles(dir: string, depth = 0): FileItem[] {
  if (depth > 3) return [];
  const SKIP = new Set(['.git', 'node_modules', '__pycache__', 'venv', '.venv', 'dist', 'build']);
  try {
    return readdirSync(dir).flatMap(name => {
      if (SKIP.has(name)) return [];
      const full = join(dir, name);
      const stat = statSync(full);
      const path = relative(process.cwd(), full);
      if (stat.isDirectory()) {
        return [{ path, type: 'directory' as const }, ...scanLocalFiles(full, depth + 1)];
      }
      return [{ path, type: 'file' as const }];
    });
  } catch {
    return [];
  }
}
```

Use this in `useEffect` for `triggerMode === 'file'` instead of `api.scanWorkspace()`:

```typescript
useEffect(() => {
  if (state.triggerMode === 'file') {
    const files = scanLocalFiles(state.workspace);
    setFiles(files);
  }
}, [state.triggerMode, state.workspace]);
```

```typescript
function buildSessionMarkdown(state: AppState): string {
  const date = new Date().toLocaleString();
  const lines = [
    `# Elith Session — ${date}`,
    ``,
    `**Workspace:** ${state.workspace}`,
    `**Model:** ${state.model}`,
    `**Branch:** ${state.branch}`,
    ``,
    `---`,
    ``,
  ];

  for (const msg of state.messages) {
    if (msg.role === 'user') {
      lines.push(`**You:** ${msg.text}`, ``);
```
